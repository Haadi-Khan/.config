#+begin_src emacs-lisp :tangle config.el
(defun haadi/ghostel--buffer-name (popup-p directory)
  "Return a stable Ghostel buffer name for DIRECTORY.
When POPUP-P is non-nil, scope the buffer to the current workspace."
  (if popup-p
      (format "*doom:ghostel-popup:%s*"
              (if (bound-and-true-p persp-mode)
                  (safe-persp-name (get-current-persp))
                "main"))
    (format "*ghostel:%s*"
            (abbreviate-file-name (directory-file-name directory)))))

(defun haadi/ghostel--open (buffer-name directory)
  "Open BUFFER-NAME as a Ghostel terminal in DIRECTORY."
  (let ((default-directory (file-name-as-directory directory))
        (ghostel-buffer-name buffer-name)
        ;; Keep our vterm-like stable names instead of following OSC titles.
        (ghostel-buffer-name-function nil))
    (let ((buffer (ghostel)))
      (with-current-buffer buffer
        (setq-local ghostel-buffer-name-function nil))
      buffer)))

(defun haadi/ghostel-toggle (arg)
  "Toggle a workspace-local Ghostel popup at the project root.
With prefix ARG, kill and recreate the popup first."
  (interactive "P")
  (require 'ghostel)
  (let* ((directory (or (doom-project-root) default-directory))
         (buffer-name (haadi/ghostel--buffer-name t directory))
         (buffer (get-buffer buffer-name))
         (window (and buffer (get-buffer-window buffer))))
    (when (and arg buffer)
      (let ((confirm-kill-processes nil))
        (kill-buffer buffer))
      (setq buffer nil
            window nil))
    (cond ((window-live-p window)
           (delete-window window))
          (t
           (unless (buffer-live-p buffer)
             (setq buffer (haadi/ghostel--open buffer-name directory)))
           (pop-to-buffer buffer)))
    buffer))

(defun haadi/ghostel-here (arg)
  "Open Ghostel in this window at the project root.
With prefix ARG, use the current directory instead."
  (interactive "P")
  (require 'ghostel)
  (let* ((directory (if arg
                        default-directory
                      (or (doom-project-root) default-directory)))
         (buffer-name (haadi/ghostel--buffer-name nil directory)))
    (haadi/ghostel--open buffer-name directory)))

(use-package! ghostel

(defun haadi/org-indent-list-item-or-cycle ()
  "Indent the list item at point, or run the usual Org TAB action."
  (interactive)
  (if-let ((item (org-in-item-p)))
      (save-excursion
        (goto-char item)
        (org-indent-item-tree))
    (org-cycle)))

(defun haadi/org-outdent-list-item-or-cycle ()
  "Outdent the list item at point, or run the usual Org Shift-TAB action."
  (interactive)
  (if-let ((item (org-in-item-p)))
      (save-excursion
        (goto-char item)
        (org-outdent-item-tree))
    (org-shifttab)))

(map! :map org-mode-map
      :ni "TAB" #'haadi/org-indent-list-item-or-cycle

      :ni "<S-tab>" #'haadi/org-outdent-list-item-or-cycle
      :ni "<S-iso-lefttab>" #'haadi/org-outdent-list-item-or-cycle)

*** Latex Rendering
For some reason (even on emacs-plus) LaTeX embeds look like utter garbage on Mac (and Linux too for that matter). So, we do some elisp magic to make them look great! We render them at 3x the size, and then scale them down
#+begin_src emacs-lisp :tangle config.el
(setq org-preview-latex-default-process 'dvisvgm)
(plist-put org-format-latex-options :scale 3)
(defun my/image-scale-advice (image)
  (let* ((factor (image-property image :scale))
         (new-factor (if factor
                         (/ factor 2.0)
                       0.5)))
    (image--set-property image :scale new-factor)
    image))
(advice-add 'org--create-inline-image :filter-return #'my/image-scale-advice)
(defun my/overlay-scale-advice (beg end image &optional imagetype)
  (mapc (lambda (ov) (if (equal (overlay-get ov 'org-overlay-type) 'org-latex-overlay)
                                (overlay-put ov
                                             'display
                                             (list 'image :type (or (intern imagetype) 'png) :file image :ascent 'center :scale 0.5))))
        (overlays-at beg)))
(advice-add 'org--make-preview-overlay :after #'my/overlay-scale-advice)

;; Keep Org independent of AucTeX: the Doom LaTeX module is disabled, and
;; enabling `org-cdlatex-mode' makes every Org buffer fail to open without it.

    (add-to-list 'org-latex-packages-alist '("" "braket" t)))

    "Return the level of the heading containing point."
    (or (haadi/markdown-heading-level-at-point)
        (save-excursion
          (let ((change (previous-single-property-change
                         (line-beginning-position) 'markdown-heading
                         nil (point-min))))
            (when (> change (point-min))
              (goto-char (1- change))
              (haadi/markdown-heading-level-at-point))))))

    "Remove our virtual indentation from PREFIX."
    (if (and (stringp prefix)
             (> (length prefix) 0)
             (get-text-property 0 'haadi/markdown-indent prefix))
        (substring prefix
                   (or (next-single-property-change
                        0 'haadi/markdown-indent prefix)
                       (length prefix)))

  (defun haadi/markdown-indent-region (beg end)
    "Visually indent Markdown from BEG to END like `org-indent-mode'."
    (save-match-data
      (save-excursion
        (save-restriction
          (widen)
          (goto-char beg)
          (forward-line 0)
          (setq beg (point))
          (goto-char end)
          (unless (bolp) (forward-line 1))
          (setq end (point))
          (goto-char beg)
          (let ((level (haadi/markdown-containing-heading-level)))
            (with-silent-modifications
              (while (< (point) end)
                (let* ((heading-level (haadi/markdown-heading-level-at-point))
                       (_ (when heading-level (setq level heading-level)))
                       ;; Markdown # is the title; ## maps to Org's *,
                       ;; ### to **, and so on.
                       (columns
                        (if (not markdown-hide-markup)
                            0
                          (if heading-level
                              (* (max 0 (- level 2))
                                 (max 0 (1- haadi/markdown-indent-per-level)))
                            (* (max 0 (1- (or level 1)))
                               haadi/markdown-indent-per-level))))
                       (indent (propertize (make-string columns ?\s)
                                          'haadi/markdown-indent t))
                       (old-wrap (haadi/markdown-strip-indent-prefix
                                  (get-text-property (point) 'wrap-prefix)))
                       (next-line (line-beginning-position 2)))
                  (add-text-properties
                   (point) next-line
                   `(line-prefix ,indent
                     wrap-prefix ,(if (stringp old-wrap)
                                      (concat indent old-wrap)
                                    indent)))
                  (goto-char next-line)))))))))

    "Extend refontification through the next heading after END."

Outside an ATX heading, insert a normal Markdown-aware newline."

**** Tables
Tables open as a wrapped, bordered reading view using
[[https://github.com/dnouri/markdown-table-wrap][markdown-table-wrap-pretty]]. This is display-only: each record stays on one
physical line in the saved Markdown. Editing reveals the source; leaving Evil
Insert state renders the current table again. [[https://github.com/casouri/valign][Valign]] aligns the source in GUI
frames, including mixed fonts. Very large source tables fall back to ordinary
Markdown alignment to keep editing responsive.

- =TAB= / =S-TAB=: align and move to the next / previous cell; =RET=: next row.
- =M-arrow=: move a row / column. =M-S-up/down=: delete / insert a row;
  =M-S-left/right=: delete / insert a column. These bindings are table commands.
- =SPC m T=: table menu (insert, align, sort, transpose, convert a region,
  rows, columns, and rendering).
- =SPC m T p= or =C-c C-x C-k=: toggle the wrapped reading view. Off a table,
  toggle all tables; =C-u= forces all pretty, =C-u C-u= forces all source.
- =SPC m T e= or =C-c `=: edit a long cell in a separate, soft-wrapped buffer.
  =C-c C-c= commits, =C-c C-k= cancels. Keep one logical line; use =<br>= for
  an intentional line break and =\|= for a literal pipe.
- =SPC m T v=: toggle pixel alignment; =SPC m T w=: toggle line truncation
  when working directly on a wide source table.

Org spreadsheet formulas and merged cells are not part of Markdown pipe-table
syntax. This setup retains Markdown's native table editor and file format.

  (defun haadi/markdown-table-cell-bounds ()
    "Return the unpadded cell bounds at point, respecting escaped pipes."
    (unless (and (markdown-table-at-point-p)
                 (not (markdown-table-hline-at-point-p)))
      (user-error "Place point in a table data or header cell"))
    (let ((pos (point)) pipes left-pipes beg end)
      (save-excursion
        (beginning-of-line)
        (while (search-forward "|" (line-end-position) t)
          (let ((pipe (1- (point))))
            (unless (or (eq (char-before pipe) ?\\)
                        (markdown--thing-at-wiki-link pipe))
              (push pipe pipes))))
        (setq pipes (nreverse pipes))
        (unless pipes (user-error "Table needs pipe delimiters"))
        (setq left-pipes
              (if (string-blank-p
                   (buffer-substring-no-properties
                    (1+ (car (last pipes))) (line-end-position)))
                  (butlast pipes)
                pipes))
        ;; At a boundary, choose the cell to its right; at the last
        ;; closing pipe, choose the final cell instead.
        (setq beg (or (car (last (seq-filter (lambda (p) (<= p pos))
                                           left-pipes)))
                      (car pipes))
              end (or (seq-find (lambda (p) (> p beg)) pipes)
                      (line-end-position)))
        (goto-char (1+ beg))
        (skip-chars-forward " \t" end)
        (setq beg (point))
        (goto-char end)
        (skip-chars-backward " \t" beg)
        (cons beg (point)))))
